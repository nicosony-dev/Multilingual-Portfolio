<script>

(function () {

  // ===== Teacher: paste your Web app URL (ends in /exec) between the quotes =====

  var WEB_APP_URL = 'https://script.google.com/a/macros/district87.org/s/AKfycbxDeKitjUE1Sp9e6pPnM8oiGwedTo7oFXTo9r9Xg61z8RhOeYxFjFssMPNMh9r4Rzgxsw/exec';

  // Longest recording, in seconds

  var MAX_SECONDS = 60;

  // ==============================================================================


  var FACES = ['🦊','🐼','🐸','🦁','🐯','🐵','🐰','🐻','🐨','🐷','🐙','🦄','🐢','🦉','🐬','🦋','🐞','🐝'];

  var classCode = new URLSearchParams(location.search).get('c') || '';


  var $ = function (id) { return document.getElementById(id); };

  var screens = ['loading','names','record','review','sending','done','error'];

  function show(name) {

    screens.forEach(function (s) { $('s-' + s).classList.toggle('active', s === name); });

  }


  var state = { student: '', prompt: null, blob: null, mime: '', stream: null, recorder: null, chunks: [], tick: null, secs: 0, previewUrl: null };


  function fail(title, msg, backTo) {

    $('err-title').textContent = title;

    $('err-msg').textContent = msg;

    $('btn-err').onclick = function () { show(backTo || 'names'); };

    show('error');

  }


  /* ---------- load roster ---------- */

  function loadRoster() {

    show('loading');

    if (!classCode) return fail('Ask your teacher', 'This page needs a class link.', 'loading');

    fetch(WEB_APP_URL + '?action=roster&c=' + encodeURIComponent(classCode))

      .then(function (r) { return r.json(); })

      .then(function (data) {

        if (!data.ok) return fail('Ask your teacher', 'This class link is not working.', 'loading');

        state.prompt = data.prompt;

        var box = $('names');

        box.innerHTML = '';

        data.students.forEach(function (name, i) {

          var b = document.createElement('button');

          b.className = 'name';

          var face = document.createElement('span');

          face.className = 'face';

          face.textContent = FACES[i % FACES.length];

          var label = document.createElement('span');

          label.textContent = name;

          b.appendChild(face); b.appendChild(label);

          b.onclick = function () { pickStudent(name); };

          box.appendChild(b);

        });

        show('names');

      })

      .catch(function () {

        fail('No internet', 'Check the Wi-Fi, then tap OK to try again.', 'loading');

        $('btn-err').onclick = loadRoster;

      });

  }


  function pickStudent(name) {

    state.student = name;

    $('p-emoji').textContent = state.prompt.emoji;

    $('p-text').textContent = state.prompt.text;

    resetRecordScreen();

    show('record');

  }


  /* ---------- read the question aloud ---------- */

  $('btn-hear').onclick = function () {

    if (!('speechSynthesis' in window)) return;

    window.speechSynthesis.cancel();

    var u = new SpeechSynthesisUtterance(state.prompt.text);

    u.rate = 0.85;

    window.speechSynthesis.speak(u);

  };


  /* ---------- recording ---------- */

  function pickType() {

    if (!window.MediaRecorder) return '';

    var types = ['audio/mp4', 'audio/webm;codecs=opus', 'audio/webm', 'audio/ogg'];

    for (var i = 0; i < types.length; i++) {

      if (MediaRecorder.isTypeSupported(types[i])) return types[i];

    }

    return '';

  }


  function resetRecordScreen() {

    $('btn-rec').classList.remove('live');

    $('rec-label').textContent = 'Record';

    $('btn-rec').setAttribute('aria-label', 'Start recording');

    $('bar').style.width = '0%';

    $('btn-rec').disabled = false;

  }


  function releaseMic() {

    if (state.stream) { state.stream.getTracks().forEach(function (t) { t.stop(); }); state.stream = null; }

  }


  function startRecording() {

    if (!navigator.mediaDevices || !window.MediaRecorder) {

      return fail('Cannot record', 'This iPad needs to be updated to record. Ask your teacher.', 'record');

    }

    window.speechSynthesis && window.speechSynthesis.cancel();

    navigator.mediaDevices.getUserMedia({ audio: true }).then(function (stream) {

      state.stream = stream;

      var type = pickType();

      state.recorder = type ? new MediaRecorder(stream, { mimeType: type }) : new MediaRecorder(stream);

      state.chunks = [];

      state.recorder.ondataavailable = function (e) { if (e.data && e.data.size) state.chunks.push(e.data); };

      state.recorder.onstop = function () {

        clearInterval(state.tick);

        releaseMic();

        state.mime = state.recorder.mimeType || type || 'audio/mp4';

        state.blob = new Blob(state.chunks, { type: state.mime });

        if (state.blob.size < 1000) {

          resetRecordScreen();

          return fail('Try again', 'We did not hear anything. Tap the red button and talk.', 'record');

        }

        if (state.previewUrl) URL.revokeObjectURL(state.previewUrl);

        state.previewUrl = URL.createObjectURL(state.blob);

        $('player').src = state.previewUrl;

        $('btn-play').textContent = '▶ Listen';

        show('review');

      };

      state.recorder.start();

      state.secs = 0;

      $('btn-rec').classList.add('live');

      $('rec-label').textContent = 'Stop';

      $('btn-rec').setAttribute('aria-label', 'Stop recording');

      state.tick = setInterval(function () {

        state.secs += 0.25;

        $('bar').style.width = Math.min(100, (state.secs / MAX_SECONDS) * 100) + '%';

        if (state.secs >= MAX_SECONDS) stopRecording();

      }, 250);

    }).catch(function () {

      fail('Microphone is off', 'Tap Allow when the iPad asks to use the microphone. Ask your teacher if it does not ask.', 'record');

    });

  }


  function stopRecording() {

    if (state.recorder && state.recorder.state === 'recording') state.recorder.stop();

  }


  $('btn-rec').onclick = function () {

    if (state.recorder && state.recorder.state === 'recording') stopRecording();

    else startRecording();

  };


  $('btn-back1').onclick = function () { show('names'); };


  /* ---------- review ---------- */

  var player = $('player');

  $('btn-play').onclick = function () {

    if (player.paused) { player.currentTime = 0; player.play(); $('btn-play').textContent = '■ Stop'; }

    else { player.pause(); $('btn-play').textContent = '▶ Listen'; }

  };

  player.onended = function () { $('btn-play').textContent = '▶ Listen'; };


  $('btn-retry').onclick = function () {

    player.pause();

    state.blob = null;

    resetRecordScreen();

    show('record');

  };


  /* ---------- send ---------- */

  function toBase64(blob) {

    return new Promise(function (resolve, reject) {

      var r = new FileReader();

      r.onload = function () { resolve(String(r.result).split(',')[1]); };

      r.onerror = reject;

      r.readAsDataURL(blob);

    });

  }


  $('btn-send').onclick = function () {

    player.pause();

    show('sending');

    toBase64(state.blob).then(function (b64) {

      return fetch(WEB_APP_URL, {

        method: 'POST',

        headers: { 'Content-Type': 'text/plain;charset=utf-8' }, // avoids a browser preflight

        body: JSON.stringify({ classCode: classCode, student: state.student, mimeType: state.mime, audio: b64 })

      });

    })

    .then(function (r) { return r.json(); })

    .then(function (res) {

      if (!res.ok) throw new Error(res.error || 'failed');

      state.blob = null;

      $('done-name').textContent = 'Great job, ' + state.student + '!';

      show('done');

      setTimeout(function () { show('names'); }, 4000);

    })

    .catch(function () {

      // The recording is kept, so the child can just tap Send again.

      fail('Did not send', 'Your voice is still saved here. Check the Wi-Fi, then tap OK and send again.', 'review');

    });

  };


  loadRoster();

})();

</script>

</body>
